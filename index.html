ب. قواعد الأمان الحازمة (Firestore Security Rules - RBAC)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function: Check if authenticated
    function isAuthenticated() {
      return request.auth != null;
    }
    
    // Helper function: Verify Doctor Admin role
    function isDoctor() {
      return isAuthenticated() && 
        request.auth.token.role == 'doctor_admin' ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'doctor_admin';
    }
    
    // Helper function: Verify Patient ownership
    function isOwner(patientId) {
      return isAuthenticated() && request.auth.uid == patientId;
    }

    // Users Collection
    match /users/{userId} {
      allow read: if isAuthenticated();
      allow write: if isDoctor() || request.auth.uid == userId;
    }

    // Patients & Clinical Records
    match /patients/{patientId} {
      // Doctor can read/write any patient record; Patient can only read her own file
      allow read: if isDoctor() || isOwner(patientId);
      allow create, update, delete: if isDoctor();

      // Nested InBody measurements
      match /inbody_records/{recordId} {
        allow read: if isDoctor() || isOwner(patientId);
        allow create, update: if isDoctor() || isOwner(patientId);
        allow delete: if isDoctor();
      }

      // Nested Daily logs & Water tracker
      match /daily_logs/{dateKey} {
        allow read, write: if isDoctor() || isOwner(patientId);
      }
    }

    // Diet Plans
    match /diet_plans/{planId} {
      allow read: if isAuthenticated() && 
        (isDoctor() || resource.data.patientId == request.auth.uid);
      allow write: if isDoctor();
    }
  }
}

ب. إعدادات Service Worker (Workbox Offline-First Strategy)

// sw.js - Service Worker with Caching & Offline Strategies
const CACHE_NAME = 'dr-rama-clinic-v1';
const STATIC_ASSETS = [
  '/',
  '/index.html',
  '/manifest.json',
  '/css/tailwind.css',
  '/js/app.js',
  'https://cdn.tailwindcss.com/3.4.17',
  'https://cdn.jsdelivr.net/npm/chart.js@4.4.1/dist/chart.umd.min.js'
];

// 1. Install Event: Cache Core App Shell
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      console.log('[SW] Pre-caching offline shell');
      return cache.addAll(STATIC_ASSETS);
    })
  );
  self.skipWaiting();
});

// 2. Activate Event: Clean up stale caches
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) => {
      return Promise.all(
        keys.filter((key) => key !== CACHE_NAME).map((key) => caches.delete(key))
      );
    })
  );
  self.clients.claim();
});

// 3. Fetch Event: Network-First for API/Data, Cache-First for Assets
self.addEventListener('fetch', (event) => {
  const url = new URL(event.request.url);

  // Firestore / API calls: Network first with offline fallback
  if (url.origin.includes('firestore.googleapis.com') || url.pathname.startsWith('/api/')) {
    event.respondWith(
      fetch(event.request).catch(() => caches.match(event.request))
    );
    return;
  }

  // Static Assets: Stale-While-Revalidate
  event.respondWith(
    caches.match(event.request).then((cachedResponse) => {
      const fetchPromise = fetch(event.request).then((networkResponse) => {
        if (networkResponse && networkResponse.status === 200) {
          const responseToCache = networkResponse.clone();
          caches.open(CACHE_NAME).then((cache) => cache.put(event.request, responseToCache));
        }
        return networkResponse;
      }).catch(() => cachedResponse);

      return cachedResponse || fetchPromise;
    })
  );
});

أ. كود محرك التغذية السريرية (ClinicalNutritionEngine.ts)

/**
 * ClinicalNutritionEngine.ts
 * Implementation of Mifflin-St Jeor Clinical Equation for Females
 * Used by Dr. Rama Safklo Clinic
 */

export interface PatientBiometrics {
  ageYears: number;
  weightKg: number;
  heightCm: number;
  activityLevel: 1.2 | 1.375 | 1.55 | 1.725;
  calorieAdjustmentKcal: number; // e.g. -500 for fat loss
}

export interface MacroSplitRatios {
  proteinRatio: number; // e.g. 0.30 (30%)
  carbsRatio: number;   // e.g. 0.40 (40%)
  fatsRatio: number;    // e.g. 0.30 (30%)
}

export interface NutritionPrescription {
  bmrKcal: number;
  tdeeKcal: number;
  targetDailyCalories: number;
  proteinGrams: number;
  carbsGrams: number;
  fatsGrams: number;
  proteinGramsPerKg: number;
}

export class ClinicalNutritionEngine {
  /**
   * Calculates Basal Metabolic Rate (BMR) for Females
   * Formula: (10 * weight) + (6.25 * height) - (5 * age) - 161
   */
  public static calculateFemaleBMR(weightKg: number, heightCm: number, ageYears: number): number {
    const bmr = (10 * weightKg) + (6.25 * heightCm) - (5 * ageYears) - 161;
    return Math.round(bmr);
  }

  /**
   * Calculates Total Daily Energy Expenditure (TDEE) and Macro Grams
   */
  public static generatePlan(
    bio: PatientBiometrics,
    ratios: MacroSplitRatios = { proteinRatio: 0.30, carbsRatio: 0.40, fatsRatio: 0.30 }
  ): NutritionPrescription {
    const bmr = this.calculateFemaleBMR(bio.weightKg, bio.heightCm, bio.ageYears);
    const tdee = Math.round(bmr * bio.activityLevel);
    
    // Safety check: Never drop below BMR - 150 or absolute 1200 kcal for adult females
    let targetCals = tdee + bio.calorieAdjustmentKcal;
    if (targetCals < 1200) {
      targetCals = 1200;
    }

    // Macro energy values: Protein = 4 kcal/g, Carbs = 4 kcal/g, Fat = 9 kcal/g
    const proteinGrams = Math.round((targetCals * ratios.proteinRatio) / 4);
    const carbsGrams = Math.round((targetCals * ratios.carbsRatio) / 4);
    const fatsGrams = Math.round((targetCals * ratios.fatsRatio) / 9);

    return {
      bmrKcal: bmr,
      tdeeKcal: tdee,
      targetDailyCalories: targetCals,
      proteinGrams,
      carbsGrams,
      fatsGrams,
      proteinGramsPerKg: Number((proteinGrams / bio.weightKg).toFixed(2))
    };
  }
}

export class FoodExchangeEngine {
  /**
   * Calculates the exact equivalent portion in grams for a replacement food item
   * based on the primary macronutrient of that food category.
   */
  public static calculateEquivalentPortion(
    originalItem: FoodItem,
    originalGrams: number,
    targetItem: FoodItem
  ): { targetGrams: number; calculatedCalories: number; protein: number; carbs: number; fats: number } {
    let targetGrams: number;

    if (originalItem.category === 'lean_protein' || originalItem.category === 'dairy') {
      // Balance by protein content
      const originalProteinGrams = (originalItem.proteinPer100g * originalGrams) / 100;
      targetGrams = Math.round((originalProteinGrams / targetItem.proteinPer100g) * 100);
    } else if (originalItem.category === 'complex_carb') {
      // Balance by carbohydrate content
      const originalCarbsGrams = (originalItem.carbsPer100g * originalGrams) / 100;
      targetGrams = Math.round((originalCarbsGrams / targetItem.carbsPer100g) * 100);
    } else {
      // Balance by fat content
      const originalFatGrams = (originalItem.fatsPer100g * originalGrams) / 100;
      targetGrams = Math.round((originalFatGrams / targetItem.fatsPer100g) * 100);
    }

    return {
      targetGrams,
      calculatedCalories: Math.round((targetItem.caloriesPer100g * targetGrams) / 100),
      protein: Number(((targetItem.proteinPer100g * targetGrams) / 100).toFixed(1)),
      carbs: Number(((targetItem.carbsPer100g * targetGrams) / 100).toFixed(1)),
      fats: Number(((targetItem.fatsPer100g * targetGrams) / 100).toFixed(1))
    };
  }
}
